# SAFU_ProProgramming

НАЧАЛО
    Ввод a, b

    ПОКА a <> b
        ЕСЛИ a > b ТО
            a := a - b
        ИНАЧЕ
            b := b - a
        КОНЕЦ ЕСЛИ
    КОНЕЦ ПОКА

    Вывод a
КОНЕЦ

НАЧАЛО
    Ввод N
    S := 0
    i := 1

    ПОКА i <= N
        Ввод x
        S := S + x
        i := i + 1
    КОНЕЦ ПОКА

    Вывод S
КОНЕЦ

НАЧАЛО
    Ввод a, b, h
    x := a

    ПОКА x <= b
        ЕСЛИ x >= 6 ТО
            y := -2 * sin(x) + 4
        ИНАЧЕ
            y := 3 * cos(x) - 2
        КОНЕЦ ЕСЛИ

        Вывод x, y
        x := x + h
    КОНЕЦ ПОКА
КОНЕЦ

НАЧАЛО
    Ввод n

    e := 2
    fact := 1
    i := 2

    ПОКА i <= n
        fact := fact * i
        e := e + 1 / fact
        i := i + 1
    КОНЕЦ ПОКА

    Вывод e
КОНЕЦ

НАЧАЛО
    Ввод a, b

    Ввод x

    ПОКА x < a ИЛИ x > b
        Вывод "Повторите ввод"
        Ввод x
    КОНЕЦ ПОКА

    Вывод x
КОНЕЦ

НАЧАЛО
    Ввод eps

    S := 0
    k := 0

    ПОКА ИСТИНА
        term := (-1)^k / 2^k
        S_old := S
        S := S + term

        ЕСЛИ |S - S_old| < eps ТО
            ВЫХОД ИЗ ЦИКЛА
        КОНЕЦ ЕСЛИ

        k := k + 1
    КОНЕЦ ПОКА

    Вывод S
КОНЕЦ
